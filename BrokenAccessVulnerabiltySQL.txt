ALTER procedure [dbo].[GetUserForAuthentication]
@UserId nvarchar(20)
As
Select UserId, UserName, EmployeeCode, UP.BranchId, UP.DepartmentId, Password, EmailAddress,
	UserLevel, UserActive, B.BranchCode, B.BranchName, D.DepartmentCode, D.DepartmentName, 
	IIf(ForcePwdChange = 1 or Datediff(dd, LastPwdChangedOn, getdate()) > PasswordAge, 1, 0) as ForcePwdChange,ISNULL(UP.ForceSessionExpired,0) as ForceSessionExpired
From UserProfile UP
Inner Join Branch B on B.BranchId = UP.BranchId
Inner Join Department D on D.DepartmentId = UP.DepartmentId, Configuration G
Where UserId = @UserId;
EXEC [SaveUserUpdatedSession] @UserId

ALTER Procedure [dbo].[SaveUserOption]
@NewUserId		Varchar(20),
@OptionId		smallint,
@AllowAccess	bit,
@AllowAdd		bit,
@AllowEdit		bit,    
@UserId			varchar(20)
As
If @AllowAccess = 0 
Begin
	Delete From UserProfile_Option Where OptionId = @OptionId And UserId = @NewUserId
End       
Else If Exists(Select OptionId from UserProfile_Option Where UserId = @NewUserId And OptionId = @OptionId)
Begin
	Update UserProfile_Option
	Set IsVisible = @AllowAccess,
		CanAdd = @AllowAdd,
		CanEdit = @AllowEdit,
		UpdatedBy = @UserId,
		UpdatedOn = getdate()
	Where OptionId = @OptionId And UserId = @NewUserId
End
Else
Begin
	Insert Into UserProfile_Option
	(UserId, OptionId, IsVisible, CanAdd, CanEdit, CreatedBy)
	values(@NewUserId, @OptionId, @AllowAccess, @AllowAdd, @AllowEdit, @UserId)
End
update ARS.dbo.UserProfile set ForceSessionExpired=1 where Userid=@NewUserId
